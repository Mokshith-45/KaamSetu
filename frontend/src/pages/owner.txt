import React, { useEffect, useState } from "react";
import { supabase } from "../supabaseClient";
import { useNavigate } from "react-router-dom";
import { FiBell, FiUser } from "react-icons/fi";
import "./OwnerDashboard.css";
import { Link } from "react-router-dom";


const OwnerDashboard = () => {
  const [ownerName, setOwnerName] = useState("Owner");
  const [userEmail, setUserEmail] = useState("");
  const [staffList, setStaffList] = useState([]);
  const [shifts, setShifts] = useState([]);
  const [activeSection, setActiveSection] = useState("overview");
  const [dropdownOpen, setDropdownOpen] = useState(false);
  const [newPassword, setNewPassword] = useState("");
  const [editedName, setEditedName] = useState("");
  const navigate = useNavigate();
  const [logs, setLogs] = useState([]);

useEffect(() => {
  const fetchLogs = async () => {
    const { data, error } = await supabase
      .from("attendance_logs")
      .select("*")
      .order("created_at", { ascending: false });

    if (error) console.error(error);
    else setLogs(data);
  };

  fetchLogs();

  // Optional: poll every 10s
  const interval = setInterval(fetchLogs, 10000);
  return () => clearInterval(interval);
}, []);


  const toggleDropdown = () => setDropdownOpen(!dropdownOpen);

  useEffect(() => {
    const fetchOwnerAndStaff = async () => {
      const { data: { user } } = await supabase.auth.getUser();
      if (user) {
        setUserEmail(user.email);

        const { data: profile } = await supabase
          .from("profiles")
          .select("full_name")
          .eq("id", user.id)
          .maybeSingle();
        if (profile) {
          setOwnerName(profile.full_name);
          setEditedName(profile.full_name);
        }

        const { data: staffData } = await supabase
          .from("profiles")
          .select("id, full_name, user_type, staff_roles(role)")
          .eq("user_type", "staff");
        if (staffData) {
          const formattedStaff = staffData.map(s => ({
            id: s.id,
            name: s.full_name,
            role: s.staff_roles?.[0]?.role || "N/A",
            contact: "N/A",
            status: "Active"
          }));
          setStaffList(formattedStaff);
        }

        setShifts([
          { day: "Monday", staff: "Alice", role: "Cashier", time: "9AM-2PM" },
          { day: "Monday", staff: "Bob", role: "Stock", time: "10AM-4PM" },
        ]);
      }
    };
    fetchOwnerAndStaff();
  }, []);

  const handleLogout = async () => {
    await supabase.auth.signOut();
    navigate("/");
  };

  const handlePasswordChange = async (e) => {
    e.preventDefault();
    const { error } = await supabase.auth.updateUser({ password: newPassword });
    if (error) alert(error.message);
    else alert("Password updated successfully!");
  };

  const handleNameChange = async (e) => {
    e.preventDefault();
    const { data: { user } } = await supabase.auth.getUser();
    if (!user) return;

    const { error } = await supabase
      .from("profiles")
      .update({ full_name: editedName })
      .eq("id", user.id);

    if (error) alert(error.message);
    else {
      setOwnerName(editedName);
      alert("Name updated successfully!");
    }
  };

  const sidebarItems = [
    { id: "overview", label: "Overview" },
    { id: "shopSetup", label: "Shop Setup" },
    { id: "staffManagement", label: "Staff Management" },
    { id: "aiShiftPlanner", label: "AI Shift Planner" },
    { id: "reports", label: "Reports / Payroll" },
    { id: "settings", label: "Settings" },
  ];

  return (
    <div className="dashboard-wrapper">
      {/* Sidebar */}
      <aside className="dashboard-sidebar">
        <div className="sidebar-logo">Workforce Planner</div>
        <nav className="sidebar-nav">
          {sidebarItems.map(item => (
            <button
              key={item.id}
              className={`sidebar-btn ${activeSection === item.id ? "active" : ""}`}
              onClick={() => setActiveSection(item.id)}
            >
              {item.label}
            </button>
          ))}
        </nav>
      </aside>

      {/* Main content */}
      <div className="dashboard-main">
        {/* Topbar */}
        <header className="dashboard-topbar">
          <div className="topbar-left">
            <h2>{sidebarItems.find(i => i.id === activeSection)?.label}</h2>
          </div>
          <div className="topbar-right">
            <FiBell className="icon" />
            <div className={`profile-dropdown ${dropdownOpen ? "show" : ""}`} onClick={toggleDropdown}>
              <FiUser className="icon" />
              <span>{ownerName}</span>
              <div className="dropdown-content">
                <button onClick={() => setActiveSection("settings")}>Profile</button>
                <button onClick={() => setActiveSection("settings")}>Settings</button>
                <button onClick={handleLogout}>Logout</button>
              </div>
            </div>
          </div>
        </header>

        {/* Content Sections */}
        <main className="dashboard-content">
          {activeSection === "overview" && (
            <div className="overview-cards">
              <div className="card total-staff">
                <h3>Total Staff</h3>
                <span>{staffList.length}</span>
              </div>
              <div className="card shifts-today">
                <h3>Scheduled Shifts Today</h3>
                <span>{shifts.length}</span>
              </div>
              <div className="card pending-requests">
                <h3>Pending Requests</h3>
                <span>2</span>
              </div>
              <div className="card ai-suggestion">
                <h3>AI Suggestion</h3>
                <span>+1 Staff Needed</span>
              </div>
            </div>
          )}

          {activeSection === "shopSetup" && (
            <section className="shop-setup card-section">
              <h3>Shop Setup</h3>
              <form className="shop-form">
                <input type="text" placeholder="Shop Name" />
                <input type="text" placeholder="Business Hours" />
                <input type="text" placeholder="Roles (Comma separated)" />
                <button type="submit">Save Shop Info</button>
              </form>
            </section>
          )}

          {activeSection === "staffManagement" && (
            <section className="staff-management card-section">
              <h3>Staff Management</h3>
              <button className="add-staff-btn">+ Add Staff</button>
              <table>
                <thead>
                  <tr>
                    <th>Name</th>
                    <th>Role</th>
                    <th>Contact</th>
                    <th>Status</th>
                    <th>Actions</th>
                  </tr>
                </thead>
                <tbody>
                  {staffList.map((staff, idx) => (
                    <tr key={idx}>
                      <td>{staff.name}</td>
                      <td>{staff.role}</td>
                      <td>{staff.contact}</td>
                      <td>{staff.status}</td>
                      <td>
                        <button className="edit-btn">Edit</button>
                        <button className="remove-btn">Remove</button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </section>
          )}
          {activeSection === "aiShiftPlanner" && (
            <section className="ai-shift card-section">
              <h3>AI Shift Planner</h3>
              <button className="ai-btn">Generate Draft Schedule</button>
              <div className="shift-preview">
                {shifts.map((shift, idx) => (
                  <div key={idx} className="shift-card">
                    <strong>{shift.day}</strong>: {shift.staff} ({shift.role}) <br /> {shift.time}
                  </div>
                ))}
              </div>
            </section>
          )}

          {activeSection === "reports" && (
            <section className="reports card-section">
              <h3>Reports / Payroll Summary</h3>
              <p>Staff hours and wages will be displayed here.</p>
            </section>
          )}

          {activeSection === "settings" && (
            <section className="settings card-section">
              <h3>Profile Settings</h3>
              <form onSubmit={handleNameChange}>
                <label>Edit Name:</label>
                <input
                  type="text"
                  value={editedName}
                  onChange={(e) => setEditedName(e.target.value)}
                />
                <button type="submit">Update Name</button>
              </form>

              <p><strong>Email:</strong> {userEmail}</p>

              <form onSubmit={handlePasswordChange}>
                <label>Change Password:</label>
                <input
                  type="password"
                  placeholder="New Password"
                  value={newPassword}
                  onChange={(e) => setNewPassword(e.target.value)}
                />
                <button type="submit">Update Password</button>
              </form>
            </section>
          )}
        </main>

        {/* Footer */}
  <footer className="dashboard-footer">
  <div className="footer-left">
    © 2025 Workforce Planner | <Link to="/about">About Us</Link> | <Link to="/contact">Contact</Link>
  </div>
</footer>
      </div>
    </div>
  );
};

export default OwnerDashboard;

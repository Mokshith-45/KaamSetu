import React, { useState, useEffect } from "react";
import { supabase } from "../supabaseClient";
import "./AuthPage.css";
import { useNavigate } from "react-router-dom";

const AuthPage = () => {
  const [isSignUp, setIsSignUp] = useState(false);
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [fullName, setFullName] = useState("");
  const [userType, setUserType] = useState("owner");
  const [staffRole, setStaffRole] = useState("Cashier");
  const [message, setMessage] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [typedText, setTypedText] = useState("");
  const [textIndex, setTextIndex] = useState(0);
  const [charIndex, setCharIndex] = useState(0);

  const navigate = useNavigate();

  // Texts for typewriter loop
  const texts = [
    "Workforce Planner",
    "Smart, AI-powered scheduling for your staff."
  ];

  // Typewriter effect with loop
  useEffect(() => {
    const currentText = texts[textIndex];
    if (charIndex < currentText.length) {
      const timeout = setTimeout(() => {
        setTypedText((prev) => prev + currentText[charIndex]);
        setCharIndex(charIndex + 1);
      }, 100);
      return () => clearTimeout(timeout);
    } else {
      // After finishing a line, wait then move to next text
      const timeout = setTimeout(() => {
        setCharIndex(0);
        setTypedText("");
        setTextIndex((prev) => (prev + 1) % texts.length);
      }, 1500);
      return () => clearTimeout(timeout);
    }
  }, [charIndex, textIndex]);

  const handleSignUp = async () => {
    setMessage("Creating your account...");
    try {
      const { data, error: signUpError } = await supabase.auth.signUp({
        email,
        password,
      });
      if (signUpError) throw signUpError;
      if (!data.user) throw new Error("Failed to create user.");

      // Insert into profiles
      const { error: profileError } = await supabase.from("profiles").insert([
        { id: data.user.id, full_name: fullName, user_type: userType },
      ]);
      if (profileError) throw new Error("Failed to save profile");

      // Insert staff role if staff
      if (userType === "staff") {
        const { error: roleError } = await supabase.from("staff_roles").insert([
          { staff_id: data.user.id, role: staffRole },
        ]);
        if (roleError) throw new Error("Failed to save staff role");
      }

      setMessage("Account created! Please login.");
      setIsSignUp(false);
      setEmail(""); setPassword(""); setFullName(""); setStaffRole("Cashier");
    } catch (err) {
      setMessage(err.message || "Something went wrong.");
    }
  };

  const handleLogin = async () => {
    setMessage("Logging in...");
    try {
      const { data: loginData, error: loginError } = await supabase.auth.signInWithPassword({
        email,
        password,
      });
      if (loginError) throw new Error("Invalid email or password");

      const { data: profileData, error: profileError } = await supabase
        .from("profiles")
        .select("full_name, user_type")
        .eq("id", loginData.user.id)
        .maybeSingle();
      if (profileError || !profileData) throw new Error("Profile not found!");

      localStorage.setItem("userFullName", profileData.full_name);

      if (profileData.user_type === "owner") navigate("/owner-dashboard");
      else navigate("/staff-dashboard");
    } catch (err) {
      setMessage(err.message);
    }
  };

  return (
    <div className="auth-container">
      <div className="auth-left">
        <h1 className="typewriter">{typedText}</h1>
      </div>

      <div className="auth-right">
        <div className="auth-box">
          <h2>{isSignUp ? "Create Account" : "Login"}</h2>

          {isSignUp && (
            <>
              <input
                type="text"
                placeholder="Full Name"
                value={fullName}
                onChange={(e) => setFullName(e.target.value)}
              />
              <select
                value={userType}
                onChange={(e) => setUserType(e.target.value)}
              >
                <option value="owner">Shop Owner</option>
                <option value="staff">Staff</option>
              </select>

              {userType === "staff" && (
                <select
                  value={staffRole}
                  onChange={(e) => setStaffRole(e.target.value)}
                >
                  <option value="Cashier">Cashier</option>
                  <option value="Stock">Stock</option>
                  <option value="Deliverer">Deliverer</option>
                  <option value="Other">Other</option>
                </select>
              )}
            </>
          )}

<input
  type="email"
  name="email"
  placeholder="Email"
  value={email}
  onChange={(e) => setEmail(e.target.value)}
  autoComplete="email"
/>

          <div className="password-wrapper">
            <input
              type={showPassword ? "text" : "password"}
              placeholder="Password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
            />
            <span
              className="show-password"
              onClick={() => setShowPassword(!showPassword)}
            >
              {showPassword ? "Hide" : "Show"}
            </span>
          </div>

          <button onClick={isSignUp ? handleSignUp : handleLogin}>
            {isSignUp ? "Sign Up" : "Login"}
          </button>

          <p>
            {isSignUp ? "Already have an account?" : "New user?"}{" "}
            <span onClick={() => setIsSignUp(!isSignUp)}>
              {isSignUp ? "Login" : "Sign Up"}
            </span>
          </p>

          {message && <p className="message">{message}</p>}
        </div>
      </div>
    </div>
  );
};

export default AuthPage;
